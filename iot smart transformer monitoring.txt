function renderChart(canvasId, data, color, fillColor) {
  const canvas = document.getElementById(canvasId);
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  
  const w = canvas.clientWidth || 400;
  const h = canvas.clientHeight || 200;
  canvas.width = w;
  canvas.height = h;

  ctx.clearRect(0, 0, w, h);
  if (!data || data.length < 2) return;

  let min = Math.min(...data);
  let max = Math.max(...data);
  if (min === max) { min -= 1; max += 1; }
  const range = max - min;

  const padL = 35, padR = 10, padT = 15, padB = 25;
  const plotW = w - padL - padR;
  const plotH = h - padT - padB;

  ctx.strokeStyle = "rgba(255,255,255,0.06)";
  ctx.fillStyle = "#64748b";
  ctx.font = "10px sans-serif";

  for (let i = 0; i <= 3; i++) {
    const y = padT + (plotH / 3) * i;
    const val = (max - (range / 3) * i).toFixed(1);
    ctx.beginPath();
    ctx.moveTo(padL, y);
    ctx.lineTo(w - padR, y);
    ctx.stroke();
    ctx.fillText(val, 4, y + 3);
  }

  const points = data.map((val, idx) => {
    return {
      x: padL + (idx / (data.length - 1)) * plotW,
      y: padT + plotH - ((val - min) / range) * plotH
    };
  });

  ctx.beginPath();
  ctx.moveTo(points[0].x, points[0].y);
  for (let i = 0; i < points.length - 1; i++) {
    const xc = (points[i].x + points[i + 1].x) / 2;
    const yc = (points[i].y + points[i + 1].y) / 2;
    ctx.quadraticCurveTo(points[i].x, points[i].y, xc, yc);
  }
  ctx.lineTo(points[points.length - 1].x, points[points.length - 1].y);
  ctx.lineTo(points[points.length - 1].x, h - padB);
  ctx.lineTo(points[0].x, h - padB);
  ctx.closePath();

  const grad = ctx.createLinearGradient(0, padT, 0, h - padB);
  grad.addColorStop(0, fillColor);
  grad.addColorStop(1, "rgba(0,0,0,0)");
  ctx.fillStyle = grad;
  ctx.fill();

  ctx.beginPath();
  ctx.moveTo(points[0].x, points[0].y);
  for (let i = 0; i < points.length - 1; i++) {
    const xc = (points[i].x + points[i + 1].x) / 2;
    const yc = (points[i].y + points[i + 1].y) / 2;
    ctx.quadraticCurveTo(points[i].x, points[i].y, xc, yc);
  }
  ctx.lineTo(points[points.length - 1].x, points[points.length - 1].y);
  ctx.strokeStyle = color;
  ctx.lineWidth = 2;
  ctx.stroke();
}

// =====================================================
// SENSOR READINGS
// =====================================================
void updateSensors() {
  float t = sensors.getTempCByIndex(0);
  if (t != DEVICE_DISCONNECTED_C && !isnan(t)) {
    currentTemperature = t;
  }
  sensors.requestTemperatures();

  int adcValue = analogRead(VOLTAGE_PIN);
  float sensorVoltage = (adcValue / 4095.0) * 3.3;
  float actualVoltage = sensorVoltage * VOLTAGE_RATIO;
  if (!isnan(actualVoltage)) {
    currentVoltage = actualVoltage;
  }
}

String getTimeNow() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return "Syncing Time...";
  }
  char buffer[20];
  strftime(buffer, sizeof(buffer), "%I:%M:%S %p", &timeinfo);
  return String(buffer);
}

String getStatus(float temp, float voltage) {
  if (temp > TEMP_WARNING || voltage > VOLT_WARNING) {
    return "CRITICAL";
  }
  if (temp > TEMP_NORMAL || voltage > VOLT_NORMAL) {
    return "WARNING";
  }
  return "HEALTHY";
}

void updateIndicators() {
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(YELLOW_LED, LOW);
  digitalWrite(RED_LED, LOW);
  digitalWrite(BUZZER, LOW);

  if (currentStatus == "HEALTHY") {
    digitalWrite(GREEN_LED, HIGH);
  } else if (currentStatus == "WARNING") {
    digitalWrite(YELLOW_LED, HIGH);
  } else {
    digitalWrite(RED_LED, HIGH);
    digitalWrite(BUZZER, HIGH);
  }
}

void storeReading() {
  if (historyCount < MAX_HISTORY) {
    temperatureHistory[historyCount] = currentTemperature;
    voltageHistory[historyCount] = currentVoltage;
    historyCount++;
  } else {
    for (int i = 0; i < MAX_HISTORY - 1; i++) {
      temperatureHistory[i] = temperatureHistory[i + 1];
      voltageHistory[i] = voltageHistory[i + 1];
    }
    temperatureHistory[MAX_HISTORY - 1] = currentTemperature;
    voltageHistory[MAX_HISTORY - 1] = currentVoltage;
  }
}

// =====================================================
// WEB SERVER ROUTES
// =====================================================
void handleRoot() {
  server.sendHeader("Connection", "close");
  server.send(200, "text/html", PAGE_MAIN);
}

void handleData() {
  String json;
  json.reserve(512);

  json = "{";
  json += "\"temperature\":" + String(currentTemperature, 1) + ",";
  json += "\"voltage\":" + String(currentVoltage, 1) + ",";
  json += "\"status\":\"" + currentStatus + "\",";
  json += "\"time\":\"" + getTimeNow() + "\",";
  json += "\"count\":" + String(historyCount) + ",";
  
  json += "\"tempHistory\":[";
  for (int i = 0; i < historyCount; i++) {
    json += String(temperatureHistory[i], 1);
    if (i < historyCount - 1) json += ",";
  }
  json += "],";

  json += "\"voltHistory\":[";
  for (int i = 0; i < historyCount; i++) {
    json += String(voltageHistory[i], 1);
    if (i < historyCount - 1) json += ",";
  }
  json += "]";
  json += "}";

  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Connection", "close");
  server.send(200, "application/json", json);
}

// =====================================================
// SETUP
// =====================================================
void setup() {
  Serial.begin(115200);

  pinMode(GREEN_LED, OUTPUT);
  pinMode(YELLOW_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);

  analogReadResolution(12);

  Wire.begin(OLED_SDA, OLED_SCL);
  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("OLED NOT FOUND");
  }

  sensors.begin();
  sensors.setWaitForConversion(false);
  sensors.requestTemperatures();

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, password);

  Serial.println("\nConnecting to WiFi...");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi Connected!");
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());

  if (MDNS.begin("transformer")) {
    Serial.println("mDNS responder started: http://transformer.local");
  }

  configTime(19800, 0, "pool.ntp.org", "time.nist.gov");

  server.on("/", handleRoot);
  server.on("/data", handleData);
  server.begin();

  Serial.println("Web Server Ready!");
}

// =====================================================
// MAIN LOOP
// =====================================================
void loop() {
  server.handleClient();

  static unsigned long lastSensorRead = 0;
  if (millis() - lastSensorRead >= 1000) {
    lastSensorRead = millis();

    updateSensors();
    currentStatus = getStatus(currentTemperature, currentVoltage);

    storeReading();
    updateIndicators();

    // OLED Refresh
    display.clearDisplay();
    display.setTextColor(SSD1306_WHITE);
    display.setTextSize(1);
    display.setCursor(0, 0);
    display.println("TRANSFORMER MONITOR");
    display.setCursor(0, 14);
    display.print("Temp: ");
    display.print(currentTemperature, 1);
    display.println(" C");

    display.setCursor(0, 27);
    display.print("Volt: ");
    display.print(currentVoltage, 1);
    display.println(" V");

    display.setCursor(0, 40);
    display.print("Time: ");
    display.println(getTimeNow());

    display.setCursor(0, 53);
    display.print("Status: ");
    display.print(currentStatus);
    display.display();

    Serial.printf("Temp: %.1f C | Volt: %.1f V | Status: %s\n", 
                  currentTemperature, currentVoltage, currentStatus.c_str());
  }
}
